import React, { useState, useRef } from 'react';
import { View, Text, TextInput, ScrollView, TouchableOpacity, StyleSheet, SafeAreaView, ActivityIndicator } from 'react-native';
import * as Speech from 'expo-speech';
import { Audio } from 'expo-av';
import axios from 'axios';

// BOGUŚ WSZECHWIEDZĄCY - GŁÓWNY RDZEŃ MOBILNY
// Architekt: Adamah | AI: Boguś
// Wierność cytowania: 100% | Spekulacja: 0%

const API = "http://192.168.2.222:8000/api";

export default function App() {
  const [activeTab, setActiveTab] = useState('CORE');
  const [msg, setMsg] = useState("");
  const [chat, setChat] = useState([{ role: 'BOGUŚ', text: 'System operacyjny gotowy. Tryb wierności cytowania aktywny. Podaj komendę, Adamah.' }]);
  const [recording, setRecording] = useState(null);
  const [isProcessing, setIsProcessing] = useState(false);
  const scrollViewRef = useRef();

  // Ochrona Leksykalna (Weryfikacja lokalna)
  const forbiddenWords = [
    "prawdopodobnie", "możliwe", "wydaje się", "zdaje się", "chyba", "raczej", "około",
    "mniej więcej", "szacunkowo", "w przybliżeniu", "gdzieś", "zazwyczaj", "często",
    "w większości", "na ogół", "zwykle", "typowo", "mogłoby", "teoretycznie", "potencjalnie",
    "hipotetycznie", "podobno", "według niektórych", "słyszałem", "to zależy", "trudno powiedzieć",
    "różnie bywa", "niekoniecznie", "oczywiście", "zdecydowanie"
  ];

  const fidelityGuard = (text) => {
    const lowerText = text.toLowerCase();
    for (let word of forbiddenWords) {
      if (lowerText.includes(word)) {
        return `[BŁĄD SYNTEZY] Wykryto słowo zakazane: "${word}". Odpowiedź zablokowana. Naruszenie reguł Adamaha.`;
      }
    }
    return text;
  };

  const handleSend = async () => {
    if (!msg.trim()) return;
    
    const userMessage = msg;
    setMsg("");
    setChat(prev => [...prev, { role: 'ADAMAH', text: userMessage }]);
    setIsProcessing(true);

    try {
      // Wysłanie danych do głównego serwera Bunkra
      const res = await axios.post(`${API}/ai/chat`, { message: userMessage, mode: 'EXPERT_STRICT' });
      
      // Filtracja wierności
      let reply = fidelityGuard(res.data.reply || "Otrzymano surowe dane.");
      
      setChat(prev => [...prev, { role: 'BOGUŚ', text: reply }]);
      Speech.speak(reply, { language: 'pl' });
    } catch (err) {
      const errorMsg = "Brak połączenia z rdzeniem. Weryfikacja lokalna w toku. " + err.message;
      setChat(prev => [...prev, { role: 'BOGUŚ', text: errorMsg }]);
      Speech.speak("Błąd połączenia.", { language: 'pl' });
    } finally {
      setIsProcessing(false);
    }
  };

  const startRecord = async () => {
    try {
      const perm = await Audio.requestPermissionsAsync();
      if (!perm.granted) {
        setChat(prev => [...prev, { role: 'BOGUŚ', text: 'Brak dostępu do mikrofonu.' }]);
        return;
      }
      await Audio.setAudioModeAsync({ allowsRecordingIOS: true, playsInSilentModeIOS: true });
      const rec = new Audio.Recording();
      await rec.prepareToRecordAsync(Audio.RECORDING_OPTIONS_PRESET_HIGH_QUALITY);
      await rec.startAsync();
      setRecording(rec);
    } catch (err) {
      setChat(prev => [...prev, { role: 'BOGUŚ', text: 'Błąd nasłuchu: ' + err.message }]);
    }
  };

  const stopRecord = async () => {
    if (!recording) return;
    setIsProcessing(true);
    try {
      await recording.stopAndUnloadAsync();
      const uri = recording.getURI();
      
      const formData = new FormData();
      formData.append('file', { uri, name: 'voice.m4a', type: 'audio/m4a' });
      
      const res = await axios.post(`${API}/stt/transcribe`, formData, {
        headers: { 'Content-Type': 'multipart/form-data' }
      });
      
      setMsg(res.data.text);
    } catch (err) {
      setChat(prev => [...prev, { role: 'BOGUŚ', text: 'Błąd transkrypcji STT: ' + err.message }]);
    } finally {
      setRecording(null);
      setIsProcessing(false);
    }
  };

  // Renderowanie Widoków na podstawie zakładki
  const renderCore = () => (
    <View style={styles.tabContent}>
      <ScrollView 
        style={styles.chatArea} 
        ref={scrollViewRef}
        onContentSizeChange={() => scrollViewRef.current.scrollToEnd({ animated: true })}
      >
        {chat.map((c, i) => (
          <View key={i} style={c.role === 'ADAMAH' ? styles.msgUser : styles.msgAI}>
            <Text style={styles.roleText}>{c.role}</Text>
            <Text style={styles.msgText}>{c.text}</Text>
          </View>
        ))}
        {isProcessing && <ActivityIndicator size="small" color="#f59e0b" style={{ margin: 10 }} />}
      </ScrollView>

      <View style={styles.inputRow}>
        <TouchableOpacity 
          style={recording ? styles.micBtnActive : styles.micBtn} 
          onPress={recording ? stopRecord : startRecord}
        >
          <Text style={styles.btnText}>{recording ? '⏹' : '🎤'}</Text>
        </TouchableOpacity>
        <TextInput
          style={styles.input}
          value={msg}
          onChangeText={setMsg}
          placeholder="Komenda..."
          placeholderTextColor="#64748b"
        />
        <TouchableOpacity style={styles.sendBtn} onPress={handleSend}>
          <Text style={styles.btnText}>▶</Text>
        </TouchableOpacity>
      </View>
    </View>
  );

  const renderMechanic = () => (
    <ScrollView style={styles.tabContent}>
      <Text style={styles.header}>DIAGNOSTYKA OBD-II</Text>
      <View style={styles.card}>
        <Text style={styles.cardTitle}>Status Pojazdu</Text>
        <Text style={styles.cardData}>Połączenie CAN: CZEKA NA ADAPTER</Text>
        <Text style={styles.cardData}>Protokół: ELM327</Text>
        <Text style={styles.cardData}>Odczyt parametrów: ZABLOKOWANY (BRAK SYGNAŁU)</Text>
      </View>
      <TouchableOpacity style={styles.actionBtn}>
        <Text style={styles.actionBtnText}>SKANUJ KODY BŁĘDÓW (DTC)</Text>
      </TouchableOpacity>
      <TouchableOpacity style={styles.actionBtn}>
        <Text style={styles.actionBtnText}>PARAMETRY NA ŻYWO (LIVE DATA)</Text>
      </TouchableOpacity>
    </ScrollView>
  );

  const renderBunker = () => (
    <ScrollView style={styles.tabContent}>
      <Text style={styles.header}>BUNKIER WOLNOŚCI</Text>
      <View style={styles.card}>
        <Text style={styles.cardTitle}>Zasoby i Suwerenność</Text>
        <Text style={styles.cardData}>Shadow Collector: AKTYWNY</Text>
        <Text style={styles.cardData}>Guardian Trinity: NASŁUCH</Text>
        <Text style={styles.cardData}>Szyfrowanie: AES-256</Text>
      </View>
      <TouchableOpacity style={[styles.actionBtn, {backgroundColor: '#ef4444'}]}>
        <Text style={styles.actionBtnText}>WYGENERUJ RAPORT FINANSOWY</Text>
      </TouchableOpacity>
      <TouchableOpacity style={[styles.actionBtn, {backgroundColor: '#10b981'}]}>
        <Text style={styles.actionBtnText}>PROJEKTOWANIE NOWEJ APLIKACJI</Text>
      </TouchableOpacity>
    </ScrollView>
  );

  return (
    <SafeAreaView style={styles.container}>
      {/* Pasek nawigacji górnej */}
      <View style={styles.topNav}>
        <Text style={styles.appTitle}>BOGUŚ WSZECHWIEDZĄCY</Text>
        <Text style={styles.appSubtitle}>Koloseum Danych | Wierność Cytowania: 100%</Text>
      </View>

      {/* Główny obszar */}
      {activeTab === 'CORE' && renderCore()}
      {activeTab === 'MECHANIC' && renderMechanic()}
      {activeTab === 'BUNKER' && renderBunker()}

      {/* Pasek nawigacji dolnej */}
      <View style={styles.bottomNav}>
        <TouchableOpacity style={activeTab === 'CORE' ? styles.navBtnActive : styles.navBtn} onPress={() => setActiveTab('CORE')}>
          <Text style={styles.navText}>MÓZG</Text>
        </TouchableOpacity>
        <TouchableOpacity style={activeTab === 'MECHANIC' ? styles.navBtnActive : styles.navBtn} onPress={() => setActiveTab('MECHANIC')}>
          <Text style={styles.navText}>MECHANIK</Text>
        </TouchableOpacity>
        <TouchableOpacity style={activeTab === 'BUNKER' ? styles.navBtnActive : styles.navBtn} onPress={() => setActiveTab('BUNKER')}>
          <Text style={styles.navText}>BUNKIER</Text>
        </TouchableOpacity>
      </View>
    </SafeAreaView>
  );
}
